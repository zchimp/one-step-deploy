# 格式化分区
```shell 
# CentOS7+/RHEL默认xfs；Ubuntu/debian默认ext4
# ext4格式：
sudo mkfs.ext4 /dev/sdb1
# xfs格式：
sudo mkfs.xfs /dev/sdb1
```

# 挂载磁盘
## 临时挂载
```shell 
# 创建空挂载目录
sudo mkdir -p /mnt/data
# 挂载
sudo mount /dev/sdb1 /mnt/data
# 验证
df -h /mnt/data
```
## 永久挂载
```shell 
blkid /dev/sdb1
# 输出：UUID="xxxxxxx" TYPE="ext4/xfs"

sudo cp /etc/fstab /etc/fstab.bak

sudo vim /etc/fstab
# 文件末尾新增一行（替换自己的UUID、文件系统）
UUID=刚才查到的UUID  /mnt/data  ext4  defaults  0 2
# UUID | 挂载点 | 文件系统 | 挂载参数 | dump | fsck自检 最后一列根分区填 1，数据盘统一填 2
UUID=b15f73a1-3baf-4158-bc24-fa033a27c2e6  /mnt/data  ext4  defaults  0 2
/dev/disk/by-uuid/b15f73a1-3baf-4158-bc24-fa033a27c2e6 /disk2 ext4 defaults 0 2

# 校验配置（必做，防止开机失败）
mount -a
```


# 磁盘卸载
## 按挂载点卸载（推荐）
umount /mnt/data
## 按设备卸载
umount /dev/sdb1
