```shell
#!/bin/bash
# OTLP JSON 直连 Jaeger，无 OTel Collector
set -e

# ========== 配置区 ==========
JAEGER_HOST="127.0.0.1"
# 默认端口是4318，改映射端口
JAEGER_PORT="63612"
OTLP_URL="http://${JAEGER_HOST}:${JAEGER_PORT}/v1/traces"
SERVICE_NAME="shell-direct-otel"
SPAN_NAME="shell-task-run"
# ===========================

# 生成 OTel 标准 ID（固定长度）
TRACE_ID=$(openssl rand -hex 16)   # traceId: 32位十六进制
SPAN_ID=$(openssl rand -hex 8)    # spanId: 16位十六进制
PARENT_SPAN_ID=""

# 纳秒级时间戳（OTel 强制单位）
START_TS=$(date +%s%09N)
DURATION=180000000  # 耗时 180ms
END_TS=$((START_TS + DURATION))

# 构造标准 OTLP JSON 报文
PAYLOAD=$(cat <<EOF
{
  "resourceSpans": [
    {
      "resource": {
        "attributes": [
          {"key":"service.name","value":{"stringValue":"${SERVICE_NAME}"}},
          {"key":"telemetry.sdk.name","value":{"stringValue":"shell-otel"}},
          {"key":"telemetry.sdk.version","value":{"stringValue":"1.0"}}
        ]
      },
      "scopeSpans": [
        {
          "scope": {"name":"shell-script"},
          "spans": [
            {
              "traceId":"${TRACE_ID}",
              "spanId":"${SPAN_ID}",
              "parentSpanId":"${PARENT_SPAN_ID}",
              "name":"${SPAN_NAME}",
              "kind": 1,
              "startTimeUnixNano":"${START_TS}",
              "endTimeUnixNano":"${END_TS}",
              "attributes": [
                {"key":"host.name","value":{"stringValue":"local-server"}},
                {"key":"run.env","value":{"stringValue":"test"}}
              ],
              "status": {"code": 0}
            }
          ]
        }
      ]
    }
  ]
}
EOF
)

# 发送 OTLP 请求
echo "上报 OTLP Trace 到 Jaeger: ${OTLP_URL} with ${PAYLOAD}"
curl -s -X POST \
  -H "Content-Type: application/json" \
  -d "${PAYLOAD}" \
  -w "\nHTTP 状态码: %{http_code}\n" \
  "${OTLP_URL}"

echo -e "\n==== 上报信息 ===="
echo "TraceID: ${TRACE_ID}"
echo "SpanID:  ${SPAN_ID}"
```

```shell
#!/bin/bash
set -euo pipefail

# ==================== 配置项 ====================
# OTel Collector OTLP HTTP 地址
OTLP_URL="http://localhost:5636/v1/metrics"
# 服务名、模拟指标名
SERVICE_NAME="shell-demo-service"
METRIC_NAME="shell_simulate_system_usage"
METRIC_DESC="Shell 模拟系统瞬时使用率指标"
# 生成 0~100 随机值
METRIC_VALUE=$((RANDOM % 100))
# 纳秒级时间戳（OTLP 强制要求）
TIME_NS=$(date +%s%09N)
# ================================================

# 构造标准 OTLP Metrics JSON 载荷
PAYLOAD=$(cat <<JSON
{
  "resourceMetrics": [
    {
      "resource": {
        "attributes": [
          {"key": "service.name", "value": {"stringValue": "${SERVICE_NAME}"}},
          {"key": "env", "value": {"stringValue": "test"}}
        ]
      },
      "scopeMetrics": [
        {
          "scope": {"name": "shell-metric-exporter"},
          "metrics": [
            {
              "name": "${METRIC_NAME}",
              "description": "${METRIC_DESC}",
              "unit": "%",
              "gauge": {
                "dataPoints": [
                  {
                    "timeUnixNano": "${TIME_NS}",
                    "asDouble": ${METRIC_VALUE},
                    "attributes": [
                      {"key": "node", "value": {"stringValue": "local-shell"}}
                    ]
                  }
                ]
              }
            }
          ]
        }
      ]
    }
  ]
}
JSON
)

# 发送请求
echo "[$(date)] 开始上报指标 ${METRIC_NAME}，值：${METRIC_VALUE}, payload: ${PAYLOAD}"
curl -s -X POST \
  -H "Content-Type: application/json" \
  -d "${PAYLOAD}" \
  -w "响应码: %{http_code}\n" \
  "${OTLP_URL}"
```
