# 工程管理和依赖管理
## uv
#### 安装
```shell
# windows
powershell -ExecutionPolicy ByPass -c "irm https://astral.sh/uv/install.ps1 | iex"
uv self version

# mac or linux
curl -LsSf https://astral.sh/uv/install.sh | sh
source $HOME/.local/bin/env
```
```shell
# 初始化项目	
uv init
# 添加依赖	
uv add langchain langchain-core langgraph langchain-openai python-dotenv
# 添加开发依赖	
uv add --dev ruff
# 同步环境	
uv sync
# 运行程序	
uv run python xxx.py
# 删除依赖	
uv remove xxx
# 查看依赖树	
uv tree
# 升级 uv 本体	
uv self update
```
#### uv项目初始化脚本
```shell
#!/bin/bash
set -e

# 使用方式：
# 1. chmod +x create_uv.sh
# 2. ./create_uv.sh my_project
# 不传参数默认项目名 hello_demo

PROJECT_NAME=${1:-hello_demo}

# 判断：同时存在小写[a-z] 并且 存在大写[A-Z]
if [[ "$PROJECT_NAME" =~ [A-Z] ]]; then
    echo "ERROR: 参数「$PROJECT_NAME」不能包含大写字母" >&2
    exit 1
fi

# uv会自动将项目名称的下划线转成横杠
RUN_NAME="${PROJECT_NAME//_/-}"

echo "===== 创建uv项目：${PROJECT_NAME}, runname ${RUN_NAME} ====="
uv init --name "${PROJECT_NAME}" "${PROJECT_NAME}"

cd "${PROJECT_NAME}" || exit 1

# 添加项目清华源
cat > uv.toml <<'EOF'
[[index]]
url = "https://pypi.tuna.tsinghua.edu.cn/simple"
default = true
trust = true
EOF

# 写入 Hello World 代码
mkdir -p src/${PROJECT_NAME}
cat <<EOF > src/${PROJECT_NAME}/__init__.py
def main():
    print("Hello World!")

if __name__ == "__main__":
    main()
EOF

echo "===== 执行 uv sync ====="
uv sync

echo "===== 运行测试 ====="
uv run ${RUN_NAME}

echo -e "\n✅ 项目创建完成！"
echo "进入项目目录：cd ${PROJECT_NAME}"
```

## Poetry
#### 安装
```shell
# windows
## 1.安装pipx
py -m pip install --user pipx
## 自动加入PATH
py -m pipx ensurepath
## 重启终端后执行
pipx install poetry

# linux
## 下载自动安装脚本
curl -sSL https://install.python-poetry.org | python3 -
## 添加到环境变量
echo 'export PATH="$HOME/.local/share/pypoetry/venv/bin:$PATH"' >> ~/.bashrc
source ~/.bashrc
```

#### 配置
```shell
# 效果：项目内出现 .venv 文件夹，方便IDE识别
poetry config virtualenvs.in-project true

# 项目级别：添加清华源并设为默认，每次新建项目换下
poetry source add --priority=primary tsinghua https://pypi.tuna.tsinghua.edu.cn/simple

poetry config --list
```

#### 使用
```shell
# 进入项目目录初始化
poetry init

# 添加生产依赖（自动写入 pyproject.toml）
poetry add langchain langgraph langchain-ollama python-dotenv pydantic

# 添加开发依赖（调试、格式化、测试，不会打包进生产）
poetry add ruff debugpy pytest --group dev

# 拉代码后还原环境
poetry install

# 对齐环境（安装声明依赖 + 删除多余包，最接近 go mod tidy）
poetry sync

# 运行模块（适配你 python -m python_learning.main）
poetry run python -m python_learning.main

# 退出虚拟环境/查看环境
poetry env info
```

#### 案例
```shell
# 创建项目，项目名 hello_demo
poetry new hello_demo
# hello_demo/
# ├── pyproject.toml
# ├── README.md
# ├── src
# │   └── hello_demo
# │       └── __init__.py
# └── tests

cd hello_demo

# 写一个hello world
cat <<EOF | sudo tee src/hello_demo/__init__.py
def main():
    print("Hello World!")

if __name__ == "__main__":
    main()
EOF

# 注册启动脚本
cat >> pyproject.toml <<'EOF'

[tool.poetry.scripts]
hello = "hello_demo:main"
EOF

# 执行安装生效
poetry install

# 运行
poetry run hello
```

## 项目初始化脚本
```shell
#!/bin/bash
set -e

# 使用方式：
# 1. chmod +x create_poetry.sh
# 2. ./create_poetry.sh my_project
# 不传参数默认项目名 hello_demo

PROJECT_NAME=${1:-hello_demo}

echo "===== 创建Poetry项目：${PROJECT_NAME} ====="
poetry new "${PROJECT_NAME}"

cd "${PROJECT_NAME}" || exit 1

# 写入 Hello World 代码
cat <<EOF > src/${PROJECT_NAME}/__init__.py
def main():
    print("Hello World!")

if __name__ == "__main__":
    main()
EOF

# 注册命令入口，防止重复添加
if ! grep -q "\[tool.poetry.scripts\]" pyproject.toml; then
cat >> pyproject.toml <<EOF

[tool.poetry.scripts]
hello = "${PROJECT_NAME}:main"
EOF
fi

echo "===== 执行 poetry install ====="
poetry install

echo "===== 运行测试 ====="
poetry run hello

echo -e "\n✅ 项目创建完成！"
echo "进入项目目录：cd ${PROJECT_NAME}"
```
